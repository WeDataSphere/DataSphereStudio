## 步骤2. 学习批量勾选按钮逻辑代码
1.批量勾选，常规用法如下
```vue
<template>
    <div class="page-content">
        <BTablePage :isLoading="isLoading" :actionType="actionType" :loadingText="loadingText">
            <template v-slot:search>
                ...
            </template>
            <template v-slot:operate>
                <FSpace v-if="!showSeleccotCol" size="middle">
                    <FButton @click="clickBatch('execute')">{{$t('common.batchExecute')}}</FButton>
                    <FButton @click="clickBatch('rerun')">{{$t('common.batchRerun')}}</FButton>
                </FSpace>
                <FSpace v-else>
                    <FButton type="primary" @click="confirmBathOperate">
                        {{curBatchType === 'execute' ? $t('_.确认批量执行') : $t('_.确认批量重跑')}}
                    </FButton>
                    <FButton @click="cancelBathOperate">{{$t('_.取消')}}</FButton>
                </FSpace>
            </template>
            <template v-slot:table>
                <f-table ref="tableRef" v-model:checkedKeys="checkedKeys" row-key="id" :data="wtssList" @selectionChange="handleSelectionChange">
                    <f-table-column
                        :visible="showSeleccotCol"
                        type="selection"
                        :selectable="selectable"
                        :width="32">
                    </f-table-column>
                    <f-table-column prop="execId" :label="$t('common.executionId')" :width="130" ellipsis :formatter="formatterEmptyValue" />
                    <f-table-column prop="projectName" :label="$t('common.project')" :width="200" ellipsis :formatter="formatterEmptyValue" />
                    <f-table-column prop="flowName" :label="$t('common.workFlow')" :width="200" ellipsis :formatter="formatterEmptyValue" />
                </f-table>
            </template>
            <template v-slot:pagination>
                ...
            </template>
        </BTablePage>
    </div>
</template>

<script setup>
import { ref, onMounted } from 'vue';
import {
    FButton, FInput, FSelect, FRangePicker, FSpace, FEllipsis, FMessage, FCard,
} from '@fesjs/fes-design';
import { BTablePage, BSearch, BPageLoading } from '@fesjs/traction-widget';
import { useI18n } from '@fesjs/fes';
import { formatterEmptyValue, loadingText, formatTableDatetime } from '@/common/utils';
import dataInfo from './hooks/useDataList.js';
import {
    getWtssFlowInstances, executeWtssFlowInstances, batchExecuteWtssFlowInstances, dataValidWtssFlowInstances,
} from './api';

const { t: $t } = useI18n();
// 批量执行、批量重跑
const showSeleccotCol = ref(false);
const checkedKeys = ref([]);// 表选中项
const multipleSelection = ref([]);// 表格选择项
const curBatchType = ref('');
const clickBatch = (batchType) => {
    curBatchType.value = batchType;
    showSeleccotCol.value = true;
};
const handleSelectionChange = (val) => {
    const selectTable = [];
    wtssList.value.forEach((item) => {
        if (checkedKeys.value.includes(item.id)) {
            selectTable.push(item);
        }
    });
    multipleSelection.value = selectTable;
};
const cancelBathOperate = () => {
    showSeleccotCol.value = false;
    multipleSelection.value = [];
    checkedKeys.value = [];
};
const handleBatchOperate = async (data) => {
    try {
        const params = {
            flow_instance_ids: data,
            execType: curBatchType.value,
        };
        await batchExecuteWtssFlowInstances(params);
        if (curBatchType.value === 'execute') {
            FMessage.success($t('_.批量执行已提交'));
        } else {
            FMessage.success($t('_.批量重跑已提交'));
        }
        return true;
    } catch (error) {
        console.log('error:', error);
        return false;
    }
};
const confirmBathOperate = async () => {
    if (checkedKeys.value.length === 0) {
        FMessage.warning($t('_.请选择操作数据'));
        return;
    }
    if (checkedKeys.value.length > 50) {
        FMessage.warning($t('_.操作数据不能超过50条'));
        return;
    }
    const r = await handleBatchOperate(checkedKeys.value);
    if (r) {
        cancelBathOperate();
    }
    loadListData();
};
const selectable = ({ row }) => {
    const operateList = computeOperate(row.status);
    if (curBatchType.value === 'rerun') {
        return operateList.some(item => item.value === 'rerun' || item.value === 'reTry');
    }
    return operateList.some(item => item.value === curBatchType.value);
};
</script>

```
2. 请结合用户给的任务要求，学习示例代码，在指定文件实现批量勾选功能。如果指定文件中有使用traction widget的BTablePage组件，则将批量按钮放在对应插槽v-slot:operate中，如果用户没有使用这个组件，则根据代码情况，放在表格组件的上方。

注意：如果用户未给出能否勾选的逻辑条件，则对应selectable函数也声明，直接返回true。否则实现相关逻辑。如果用户没有给出数量限制，则设置勾选数量要大于0，小于等于100。
